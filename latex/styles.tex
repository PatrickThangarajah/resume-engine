% =============================================================================
%  styles.tex: fonts and sizes shared by the resume AND the cover letter.
%  Change a size here and both documents change together.
% =============================================================================

% --- text styles (override awesome-cv defaults) --------------------------------
%  Body text at 10pt regular weight: recruiters skim, and 9pt light is hard to read.
\newcommand{\BodySize}{10pt}
\renewcommand*{\descriptionstyle}[1]{{\fontsize{\BodySize}{1.05em}\bodyfont\upshape\color{text} #1}}
\renewcommand*{\paragraphstyle}{\fontsize{\BodySize}{1.05em}\bodyfont\upshape\color{text}}
\renewcommand*{\skillsetstyle}[1]{{\fontsize{\BodySize}{1.05em}\bodyfont\color{text} #1}}

%  Three sizes only: name, section titles, everything else at body size.
\newcommand{\NameSize}{18pt}
\newcommand{\HeadSize}{11pt}
\newcommand{\SmallSize}{9pt}   % contact line and citations: long lines, kept one step smaller
\makeatletter
\renewcommand*{\headerfirstnamestyle}[1]{{\fontsize{\NameSize}{1em}\headerfont\color{text} #1}}
\renewcommand*{\headerlastnamestyle}[1]{{\fontsize{\NameSize}{1em}\headerfont\color{text} #1}}
\renewcommand*{\headerpositionstyle}[1]{{\fontsize{\BodySize}{1em}\bodyfont\scshape\color{awesome} #1}}
\renewcommand*{\headeraddressstyle}[1]{{\fontsize{\SmallSize}{1em}\headerfont\itshape\color{lighttext} #1}}
\renewcommand*{\headersocialstyle}[1]{{\fontsize{\SmallSize}{1em}\headerfont\color{text} #1}}
\renewcommand*{\sectionstyle}[1]{{\fontsize{\HeadSize}{1em}\bodyfont\bfseries\color{text}\@sectioncolor #1}}
\renewcommand*{\entrypositionstyle}[1]{{\fontsize{\BodySize}{1em}\bodyfont\color{graytext} #1}}
\renewcommand*{\entrydatestyle}[1]{{\fontsize{\BodySize}{1em}\bodyfontlight\slshape\color{graytext} #1}}
\renewcommand*{\entrylocationstyle}[1]{{\fontsize{\BodySize}{1em}\bodyfontlight\slshape\color{awesome} #1}}
\makeatother
